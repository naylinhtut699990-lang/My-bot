import os
import random
import asyncio
from telegram import Update
from telegram.ext import ApplicationBuilder, MessageHandler, filters, ContextTypes

# 1. Config Keys (သင့်ရဲ့ Bot Token ကို ဒီနေရာမှာ ထည့်ပါ)
TELEGRAM_BOT_TOKEN = "YOUR_TELEGRAM_BOT_TOKEN_HERE"

# တက်ကြွနေဆဲ Spam လုပ်ငန်းစဉ်များကို မှတ်သားရန် Dictionary (chat_id: task)
active_spams = {}

# သင်ကိုယ်တိုင် Telegram ထဲမှာ မှတ်ခိုင်းထားမည့် ဆဲစာ/ချေပစာ Database
CUSTOM_INSULTS = [
    "မင်းမှာ ဦးနှောက်ပါရဲ့လား? စကားပြောတာ အဆင့်မရှိလိုက်တာ!",
    "ငါ့လာမရှုပ်နဲ့၊ မင်းအဆင့်နဲ့ ငါ့ကို လာယှဉ်လို့ မရဘူး!",
    "တကယ်ပဲ ကလေးကလား စကားတွေ လာပြောနေတယ်၊ သွား ကစားစရာရှိတာ သွားကစား!",
    "မင်းပြောတဲ့ စာတွေကို ကြည့်ပြီး အလကားကောင်ပဲ!"
]

async def spam_loop(context: ContextTypes.DEFAULT_TYPE, chat_id: int, target_name: str):
    """ 'mt' ဖြင့် စတင်ခိုင်းပြီး ရပ်ခိုင်းသည်အထိ ဆက်တိုက် Spam မည့် Loop """
    try:
        while True:
            if not CUSTOM_INSULTS:
                break
                
            custom_pick = random.choice(CUSTOM_INSULTS)
            await context.bot.send_message(chat_id=chat_id, text=f"{target_name} {custom_pick}")
            await asyncio.sleep(2) # ၂ စက္ကန့်ခြားပြီး ဆက်တိုက် ပို့မည်
            
    except asyncio.CancelledError:
        pass

async def handle_message(update: Update, context: ContextTypes.DEFAULT_TYPE):
    if not update.message or not update.message.text:
        return

    user_text = update.message.text
    chat_id = update.message.chat_id
    bot_username = context.bot.username

    is_tagged = f"@{bot_username}" in user_text
    is_reply = update.message.reply_to_message and update.message.reply_to_message.from_user.id == context.bot.id

    if is_tagged or is_reply:
        clean_text = user_text.replace(f"@{bot_username}", "").strip()
        lower_text = clean_text.lower()
        
        # ၁။ ရပ်ခိုင်းသည့် အမိန့် ("ရပ်", "တော်ပြီ", "stop", "ငြိမ်")
        if any(keyword in lower_text for keyword in ["ရပ်", "တော်ပြီ", "stop", "ငြိမ်"]):
            if chat_id in active_spams:
                active_spams[chat_id].cancel()
                del active_spams[chat_id]
                await update.message.reply_text("🛑 ကဲ... ဆဲတာ ရပ်လိုက်ပြီကွာ။")
            else:
                await update.message.reply_text("လက်ရှိ ဆဲနေတာ ဘာမှ မရှိပါဘူး။")
            return

        # ၂။ စာအသစ် မှတ်ခိုင်းခြင်း (ဥပမာ: "မှတ်ထား: ...")
        if "မှတ်ထား:" in clean_text or "မှတ်ထား -" in clean_text:
            parts = clean_text.split(":", 1) if ":" in clean_text else clean_text.split("-", 1)
            if len(parts) > 1:
                new_insult = parts[1].strip()
                if new_insult:
                    CUSTOM_INSULTS.append(new_insult)
                    await update.message.reply_text(f"✅ ဒီဆဲစာကို မှတ်ထားလိုက်ပြီ! (စုစုပေါင်း: {len(CUSTOM_INSULTS)} ခု)")
                    return

        # ၃။ 'mt' ဖြင့် စတင်ခိုင်းပြီး ဆက်တိုက် စပန်းဆဲခိုင်းခြင်း
        if lower_text.startswith("mt") or " mt " in lower_text:
            if chat_id in active_spams:
                await update.message.reply_text("⚠️ လက်ရှိ တစ်ယောက်ကို ဆဲနေတုန်းပါ! ရပ်ခိုင်းချင်ရင် 'ရပ်' လို့ ရိုက်ပေးပါ။")
                return

            target_name = ""
            if update.message.reply_to_message:
                target_name = f"@{update.message.reply_to_message.from_user.username}" if update.message.reply_to_message.from_user.username else update.message.reply_to_message.from_user.first_name
            else:
                words = clean_text.split()
                for w in words:
                    if w.startswith("@") and w != f"@{bot_username}":
                        target_name = w
                        break
            
            if not target_name:
                target_name = "ဟိုကောင်"

            # Background Loop စတင်ခြင်း
            task = asyncio.create_task(spam_loop(context, chat_id, target_name))
            active_spams[chat_id] = task
            await update.message.reply_text(f"🔥 {target_name} ကို 'mt' စနစ်နဲ့ ရပ်ခိုင်းတဲ့အထိ စပန်းဝင်ဆဲနေပြီနော်!")

if __name__ == '__main__':
    app = ApplicationBuilder().token(TELEGRAM_BOT_TOKEN).build()
    app.add_handler(MessageHandler(filters.TEXT & (~filters.COMMAND), handle_message))
    
    print("Mt Spam Bot running...")
    app.run_polling()
